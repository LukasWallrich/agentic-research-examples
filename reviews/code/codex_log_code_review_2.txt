# Estimators applied identically to simulated and observed data.
# y is log(estimate / true value); x is log(anchor / true value).

suppressPackageStartupMessages({ library(lme4); library(metafor); library(data.table) })

# Gross-error rule: drop responses further from the true value (log scale) than both a factor of
# 1,000 and ten times the most extreme anchor used for that cell. A response equal to any anchor
# is always retained, so the rule cannot remove strongly anchored answers.
drop_gross_errors <- function(dat) {
  dat <- dat[is.finite(y)]
  dat[, y_limit := pmax(log(1000), max(abs(x)) + log(10)), by = cell]
  dat[abs(y) <= y_limit][, y_limit := NULL][]
}

# Within-item response function: y = a_cell + b x + g x|x|, with uncorrelated random intercepts,
# slopes and curvatures by cell, and random intercepts by participant. b is the average elasticity at the true value; g > 0 means the
# local elasticity rises with anchor extremity, g < 0 means it falls.
fit_within <- function(dat) {
  dat <- drop_gross_errors(dat)
  dat[, xx := x * abs(x)]
  m <- lmer(y ~ x + xx + (1 + x + xx || cell) + (1 | pid), data = dat,
            control = lmerControl(calc.derivs = FALSE, check.conv.singular = "ignore"))
  co <- summary(m)$coefficients
  b <- co["x", "Estimate"]; g <- co["xx", "Estimate"]
  V <- as.matrix(vcov(m))[c("x", "xx"), c("x", "xx")]
  # Relative curvature rho = g / b, with a delta-method standard error.
  grad <- c(-g / b^2, 1 / b)
  list(model = m, b = b, b_se = co["x", "Std. Error"], g = g, g_se = co["xx", "Std. Error"],
       rho = g / b, rho_se = sqrt(drop(t(grad) %*% V %*% grad)), n = nrow(dat))
}

# Across-cell anchoring index: log-scale index per two-anchor cell, meta-regressed on extremity
# (mean absolute log distance of the two anchors from the true value).
cell_indices <- function(rows, cells) {
  r <- merge(drop_gross_errors(rows)[, .(cell, anchorhigh, y)], cells[, .(cell, study, x_low, x_high, extremity)], by = "cell")
  r[, .(med_high = median(y[anchorhigh == 1]), med_low = median(y[anchorhigh == 0]),
        mad_high = mad(y[anchorhigh == 1]), mad_low = mad(y[anchorhigh == 0]),
        n_high = sum(anchorhigh == 1), n_low = sum(anchorhigh == 0),
        study = study[1], x_low = x_low[1], x_high = x_high[1], extremity = extremity[1]), by = cell][
    , `:=`(ai = (med_high - med_low) / (x_high - x_low),
           ai_se = 1.2533 * sqrt(pmax(mad_high, 1e-3)^2 / n_high + pmax(mad_low, 1e-3)^2 / n_low) / (x_high - x_low))][]
}

fit_across <- function(ci) {
  m <- rma.mv(ai, ai_se^2, mods = ~ extremity, random = ~ 1 | study / cell, data = ci, method = "REML")
  list(model = m, slope = unname(coef(m)["extremity"]), slope_se = unname(m$se[2]),
       intercept = unname(coef(m)["intrcpt"]), mean_ai = weighted.mean(ci$ai, 1 / ci$ai_se^2))
}

===== R/models.R =====
# Generative process models of anchored estimation on the log scale.
# All quantities are log(value / true value); sigma is the item's belief uncertainty,
# and every model parameter is expressed in units of sigma, so the models are scale-free.
#
# Each participant-item row has a belief centre mu = bias + person offset + belief noise.
# Response = model output + response noise; a small share of responses is contaminated
# (unit errors, typing slips) to mimic real estimation data.

belief_sd_share <- 0.6     # SD of belief centres across people, in sigma units
response_sd_share <- 0.3   # response noise SD, in sigma units
contamination_p <- 0.02    # share of responses with gross errors
contamination_sd <- 2      # SD (log units) of gross errors

model_names <- c(M0 = "No anchoring", M1 = "Anchor-and-adjust, fixed number of adjustments",
                 M1R = "Resource-rational anchor-and-adjust (adaptive number of adjustments)",
                 M2 = "Constant-weight integration", M3 = "Adjust to plausible boundary",
                 M4 = "Plausibility-discounted integration")

# Metropolis-Hastings adjustment from the anchor towards the belief N(mu, sigma^2)
# (Lieder et al., 2018a). steps can differ by row; proposals have SD s * sigma.
adjust <- function(x, mu, sigma, s, steps) {
  z <- x
  for (i in seq_len(max(steps))) {
    active <- steps >= i
    prop <- z + rnorm(length(z), 0, s * sigma)
    log_ratio <- ((z - mu)^2 - (prop - mu)^2) / (2 * sigma^2)
    z <- ifelse(active & log(runif(length(z))) < log_ratio, prop, z)
  }
  z
}

# Resource-rational number of adjustments: t* minimises the expected absolute error under the
# agent's own belief plus a time cost c per adjustment (Lieder et al., 2018a, Eq. 2). The expected
# error after t steps from a standardised start distance z is tabulated once per step size s.
rr_z_grid <- seq(0, 12, by = 0.1)
rr_max_t <- 300
rr_error_curves <- local({
  cache <- list()
  function(s) {
    key <- as.character(s)
    if (is.null(cache[[key]])) {
      n <- 3000
      err <- matrix(NA_real_, length(rr_z_grid), rr_max_t + 1)
      abs_err <- function(y) 2 * dnorm(y) + y * (2 * pnorm(y) - 1)   # E|y - X|, X ~ N(0, 1)
      # Fixed seed that restores the caller's random-number state afterwards
      withr::with_seed(424242, for (j in seq_along(rr_z_grid)) {
        z <- rep(rr_z_grid[j], n)
        err[j, 1] <- mean(abs_err(z))
        for (t in seq_len(rr_max_t)) {
          prop <- z + rnorm(n, 0, s)
          z <- ifelse(log(runif(n)) < (z^2 - prop^2) / 2, prop, z)
          err[j, t + 1] <- mean(abs_err(z))
        }
      })
      cache[[key]] <<- err
    }
    cache[[key]]
  }
})
rr_steps <- function(dist_sd, s, cost) {
  err <- rr_error_curves(s)
  t_star <- apply(err, 1, function(e) which.min(e + cost * (seq_along(e) - 1)) - 1)
  t_star[pmin(round(dist_sd / 0.1) + 1, length(rr_z_grid))]
}

respond <- function(model, x, mu, sigma, par) {
  d <- x - mu
  switch(model,
    M0 = mu,
    M2 = mu + par$w * d,
    M3 = ifelse(abs(d) < par$k * sigma, x, mu + sign(d) * par$k * sigma),
    M4 = mu + par$w * d * exp(-d^2 / (2 * (par$c * sigma)^2)),
    M1 = adjust(x, mu, sigma, par$s, rep(par$t, length(x))),
    M1R = {
      # kappa: the agent's own uncertainty relative to the spread of beliefs across people
      sig <- par$kappa * sigma
      adjust(x, mu, sig, par$s, rr_steps(abs(d) / sig, par$s, par$cost))
    })
}

# rows: data.table with x (log anchor / truth), cell, pid. cellpar: data.table with cell, bias, sigma.
simulate_responses <- function(model, rows, cellpar, par, person_sd = 0.15) {
  r <- merge(rows[, .(row_id = .I, cell, pid, x)], cellpar, by = "cell")
  setorder(r, row_id)
  persons <- unique(r$pid)
  u <- setNames(rnorm(length(persons), 0, person_sd), persons)
  mu <- r$bias + u[r$pid] + rnorm(nrow(r), 0, belief_sd_share * r$sigma)
  y <- respond(model, r$x, mu, r$sigma, par) + rnorm(nrow(r), 0, response_sd_share * r$sigma)
  gross <- runif(nrow(r)) < contamination_p
  y[gross] <- y[gross] + rnorm(sum(gross), 0, contamination_sd)
  y
}

# Anchoring index of each model in the Jacowitz & Kahneman (1995) design: anchors at the
# 15th and 85th percentiles of the unanchored response distribution.
jk_index <- function(model, par, n = 20000) {
  sd_unanchored <- sqrt(belief_sd_share^2 + response_sd_share^2)
  a <- qnorm(0.85) * sd_unanchored
  y_of <- function(x) {
    mu <- rnorm(n, 0, belief_sd_share)
    respond(model, rep(x, n), mu, 1, par) + rnorm(n, 0, response_sd_share)
  }
  (median(y_of(a)) - median(y_of(-a))) / (2 * a)
}

===== R/design_sets.R =====
# Inclusion rules shared by the prediction simulation and the confirmatory analysis.
# build_sets() works on design columns; when `estimate` is present it is carried along.

source(file.path("R", "read_opaq.R"))

ratio_tasktypes <- c("quantity", "number", "price", "length", "height", "distance",
                     "weight", "population", "volume", "speed", "duration", "birth rate",
                     "income", "width", "area", "circumference", "age")
# Studies whose anchors were computed from each participant's own unanchored estimates.
endogenous_anchor_studies <- c("Röseler & Wagner, 2021")

min_levels_within <- 3     # distinct anchor values per cell for the within-item set
min_log_spread <- 0.25     # across-cell set: log(A_high / A_low) of at least 0.25
min_n_side <- 10           # across-cell set: at least 10 estimates per anchor side

build_sets <- function(d) {
  d <- copy(d)
  d[, `:=`(a = suppressWarnings(as.numeric(anchor)), t = suppressWarnings(as.numeric(true_value)))]
  d[, cell := paste(reference_short, anchoring_item, sep = " || ")]
  d[, pid := paste(reference_short, participant_id, sep = " || ")]
  base <- d[anchortype %in% 1:3 & scaletype == 1 & tasktype %in% ratio_tasktypes &
              !is.na(a) & a > 0 & !is.na(t) & t > 0 &
              !reference_short %in% endogenous_anchor_studies]
  base[, t := t[1], by = cell]   # one true value per cell
  base[, x := log(a / t)]        # anchor position relative to truth (log scale)

  lev <- base[, .(n_levels = uniqueN(a), anchortype = min(anchortype)), by = cell]
  within <- base[cell %in% lev[n_levels >= min_levels_within, cell]]

  pairs <- base[anchortype %in% 2:3 & !is.na(anchorhigh)]
  pc <- pairs[, .(study = reference_short[1], item = anchoring_item[1], t = t[1],
                  n_low_values = uniqueN(a[anchorhigh == 0]), n_high_values = uniqueN(a[anchorhigh == 1]),
                  a_low = a[anchorhigh == 0][1], a_high = a[anchorhigh == 1][1],
                  n_low = sum(anchorhigh == 0), n_high = sum(anchorhigh == 1),
                  anchortype = anchortype[1]), by = cell]
  pc <- pc[n_low_values == 1 & n_high_values == 1 & a_high > a_low &
             log(a_high / a_low) >= min_log_spread & n_low >= min_n_side & n_high >= min_n_side]
  pc[, `:=`(x_low = log(a_low / t), x_high = log(a_high / t))]
  pc[, extremity := (abs(x_low) + abs(x_high)) / 2]
  across <- pairs[cell %in% pc$cell]

  list(within = within, across_rows = across, across_cells = pc)
}

# E2: per-cell OLS with separate slopes for near (|x| <= 1) and far (|x| > 1) anchors
{
  e2 <- W[, {
    h <- sign(x) * pmax(abs(x) - 1, 0)
    ok <- sum(abs(x) > 1) >= 20 && sum(abs(x) <= 1) >= 20
    if (ok) { co <- coef(summary(lm(y ~ x + h))); list(n = .N, slope_near = co["x", 1], slope_far = co["x", 1] + co["h", 1],
                                                       diff = co["h", 1], diff_se = co["h", 2]) }
    else list(n = .N, slope_near = NA_real_, slope_far = NA_real_, diff = NA_real_, diff_se = NA_real_)
  }, by = .(cell, reference_short, anchortype)]
  res$E2_cells_far_below_near <- c(est = e2[!is.na(diff), sum(diff < 0)], lo = NA, hi = e2[!is.na(diff), .N])

# E3: Hedges' g per across-set cell (log scale) against extremity, as in Röseler & Schütz (2022)
{
  e3 <- A[, {
    yh <- y[anchorhigh == 1]; yl <- y[anchorhigh == 0]
    n1 <- length(yh); n2 <- length(yl)
    sp <- sqrt(((n1 - 1) * var(yh) + (n2 - 1) * var(yl)) / (n1 + n2 - 2))
    g <- (1 - 3 / (4 * (n1 + n2) - 9)) * (mean(yh) - mean(yl)) / sp   # Hedges' g
    list(g = g, v = (n1 + n2) / (n1 * n2) + g^2 / (2 * (n1 + n2)))
  }, by = cell]
  e3 <- merge(e3, ci_A[, .(cell, study, extremity)], by = "cell")
  m3 <- rma.mv(g, v, mods = ~ extremity, random = ~ 1 | study / cell, data = e3)
  res$E3_hedges_g_slope <- c(est = m3$beta[2], lo = m3$ci.lb[2], hi = m3$ci.ub[2])
}

# E4: extremity in units of the within-condition spread of log responses (outcome-based, so
# contaminated by anchoring; shown only to probe the within/across discrepancy)
{
  sp <- A[, .(spread = sqrt((mad(y[anchorhigh == 1])^2 + mad(y[anchorhigh == 0])^2) / 2)), by = cell]
  e4 <- merge(ci_A, sp, by = "cell")[spread > 0][, ext_z := extremity / spread]
  m4 <- rma.mv(ai, ai_se^2, mods = ~ ext_z, random = ~ 1 | study / cell, data = e4)
  res$E4_slope_on_scaled_extremity <- c(est = m4$beta[2], lo = m4$ci.lb[2], hi = m4$ci.ub[2])
  m4b <- rma.mv(ai, ai_se^2, mods = ~ extremity + spread, random = ~ 1 | study / cell, data = e4)
  res$E4_slope_adjusting_spread <- c(est = m4b$beta[2], lo = m4b$ci.lb[2], hi = m4b$ci.ub[2])
  res$E4_spread_coefficient <- c(est = m4b$beta[3], lo = m4b$ci.lb[3], hi = m4b$ci.ub[3])
}

# E5: relaxed calibration. Draw each model's parameters from wide ranges (not restricted to the
# Jacowitz & Kahneman band) and ask whether any draw reproduces the observed within-item b and g
# and the across-cell d and mean index jointly (each inside the observed 95% CI).
{
  h1 <- fread(file.path("output", "confirmatory", "h1_within.csv"))[set == "pooled"]
  h2 <- fread(file.path("output", "confirmatory", "h2_across.csv"))
  sets <- build_sets(read_opaq(include_outcomes = FALSE))
  within_rows <- sets$within[, .(cell, pid, x, anchortype)]
  across_rows <- sets$across_rows[, .(cell, pid, x, anchorhigh)]
  cells <- unique(c(within_rows$cell, across_rows$cell))
  draw_par <- function(model) switch(model,
    M1 = list(s = sample(c(0.1, 0.2, 0.3, 0.5, 0.75, 1, 1.5, 2), 1), t = sample(1:60, 1)),
    M1R = list(s = sample(c(0.1, 0.2, 0.3, 0.5, 0.75, 1, 1.5, 2), 1), cost = 10^runif(1, -4, -0.5), kappa = runif(1, 0.2, 1)),
    M2 = list(w = runif(1, 0.02, 0.98)),
    M3 = list(k = runif(1, 0.05, 4)),
    M4 = list(w = runif(1, 0.1, 1), c = runif(1, 0.3, 6)))
  invisible(lapply(c(0.1, 0.2, 0.3, 0.5, 0.75, 1, 1.5, 2), rr_error_curves))   # workers inherit the cache
  jobs <- CJ(model = c("M1", "M1R", "M2", "M3", "M4"), rep = 1:300)
  e5 <- rbindlist(mclapply(seq_len(nrow(jobs)), function(i) {
    set.seed(7e5 + i)
    model <- jobs$model[i]; par <- draw_par(model)
    cellpar <- data.table(cell = cells, sigma = runif(length(cells), 0.3, 1.2), bias = rnorm(length(cells), runif(1, -0.3, 0.3), 0.3))
    w <- copy(within_rows)[, y := simulate_responses(model, within_rows, cellpar, par)]
    a <- copy(across_rows)[, y := simulate_responses(model, across_rows, cellpar, par)]

  e5[, `:=`(b_ok = b >= h1$b_lo & b <= h1$b_hi, g_ok = g >= h1$g_lo & g <= h1$g_hi,
            d_ok = d >= h2$slope_lo & d <= h2$slope_hi, ai_ok = mean_ai >= h2$mean_ai_lo & mean_ai <= h2$mean_ai_hi)]
  e5[, within_ok := b_ok & g_ok][, all_ok := within_ok & d_ok & ai_ok]
  fwrite(e5, file.path(out_dir, "E5_relaxed_calibration_draws.csv"))
  e5_sum <- e5[, .(draws = .N, within_b_and_g = sum(within_ok), across_d_and_ai = sum(d_ok & ai_ok), all_four = sum(all_ok)), by = model]
  fwrite(e5_sum, file.path(out_dir, "E5_relaxed_calibration_summary.csv"))
}

# ---- S1: raw-scale calibration ---------------------------------------------------------------
# The raw-scale index depends on the item's log uncertainty sigma, so it is averaged over the
# same sigma range the prediction simulation uses.
jk_index_raw <- function(model, par, n = 20000, sigmas = c(0.3, 0.6, 0.9, 1.2)) {
  sd_unanchored <- sqrt(belief_sd_share^2 + response_sd_share^2)
  mean(sapply(sigmas, function(sg) {
    a <- qnorm(0.85) * sd_unanchored * sg
    y_of <- function(x) {
      mu <- rnorm(n, 0, belief_sd_share * sg)
      exp(respond(model, rep(x, n), mu, sg, par) + rnorm(n, 0, response_sd_share * sg))
    }
    (median(y_of(a)) - median(y_of(-a))) / (exp(a) - exp(-a))
  }))
}

**Yes, MAJOR bugs exist.**

- **MAJOR — `R/05f_round2_checks.R`, E7:** The stated check asks whether a model matches `b`, `g`, the across-cell slope `d`, **and the mean anchoring index**.
